    Scripts de Execução Supabase SQL

    📁 supabase_data/01_schema.sql 
    -- Script 01: Criação de Tabelas e Extensões
CREATE TABLE IF NOT EXISTS public.profiles (
    id UUID PRIMARY KEY,
    username TEXT UNIQUE NOT NULL,
    password_hash TEXT NOT NULL,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.items (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    user_id UUID REFERENCES public.profiles(id) ON DELETE CASCADE NOT NULL,
    title TEXT NOT NULL,
    unit TEXT CHECK (unit IN ('GR', 'UN', 'ML')) NOT NULL,
    current_stock NUMERIC(12, 3) DEFAULT 0,
    avg_unit_cost NUMERIC(12, 4) DEFAULT 0,
    status TEXT DEFAULT 'SEM MOVIMENTAÇÕES',
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW(),
    updated_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

CREATE TABLE IF NOT EXISTS public.movements (
    id UUID PRIMARY KEY DEFAULT gen_random_uuid(),
    item_id UUID REFERENCES public.items(id) ON DELETE CASCADE NOT NULL,
    type TEXT CHECK (type IN ('IN', 'OUT')) NOT NULL,
    quantity NUMERIC(12, 3) NOT NULL,
    total_cost NUMERIC(12, 2) DEFAULT 0,
    unit_cost_calculated NUMERIC(12, 4) DEFAULT 0,
    created_at TIMESTAMP WITH TIME ZONE DEFAULT NOW()
);

    -- Script 02: Habilitação de RLS e Políticas
ALTER TABLE public.profiles ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.items ENABLE ROW LEVEL SECURITY;
ALTER TABLE public.movements ENABLE ROW LEVEL SECURITY;

-- Políticas Acesso Livre para Autenticação Simples Customizada via Tabela Profile
CREATE POLICY "Permite leitura publica perfis" ON public.profiles FOR SELECT USING (true);
CREATE POLICY "Permite insercao publica perfis" ON public.profiles FOR INSERT WITH CHECK (true);

CREATE POLICY "Permite acesso a itens por usuario" ON public.items FOR ALL USING (true);
CREATE POLICY "Permite acesso a movimentos por usuario" ON public.movements FOR ALL USING (true);
